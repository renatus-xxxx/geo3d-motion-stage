cmake_minimum_required(VERSION 3.20)
project(motion_stage_native C CXX)
include(CTest)
set(LIBV9968_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../libv9968" CACHE PATH "libv9968 source")
set(MOTION_PROFILE "8m" CACHE STRING "8m or 2m")
set(V9968_BUILD_PSG ON CACHE BOOL "Enable PSG audio" FORCE)
add_subdirectory(${LIBV9968_ROOT} libv9968)
add_executable(motion-stage src/main.c src/demo.c src/scene.c src/trig.c src/video.c src/music.c platform/pc/native.c)
option(MOTION_NATIVE_FIL "Use current FPGA R20 FIL layout" ON)
target_compile_definitions(motion-stage PRIVATE MOTION_NATIVE _CRT_SECURE_NO_WARNINGS)
if(MOTION_NATIVE_FIL)
  target_compile_definitions(motion-stage PRIVATE V9968_NATIVE_FIL)
endif()
target_include_directories(motion-stage PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/build/${MOTION_PROFILE}")
target_link_libraries(motion-stage PRIVATE geo3d v9968-window psg-audio)
psg_runtime(motion-stage)
add_custom_command(TARGET motion-stage POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/build/${MOTION_PROFILE}/motion-banks.bin" "$<TARGET_FILE_DIR:motion-stage>/motion-banks.bin")
if(WIN32 AND TARGET GLUT::GLUT)
  add_custom_command(TARGET motion-stage POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:GLUT::GLUT> $<TARGET_FILE_DIR:motion-stage>)
endif()
add_test(NAME motion-native-loop COMMAND motion-stage --ticks 4400)
set_tests_properties(motion-native-loop PROPERTIES WORKING_DIRECTORY $<TARGET_FILE_DIR:motion-stage>)

add_test(NAME motion-native-input COMMAND motion-stage --test-input)
set_tests_properties(motion-native-input PROPERTIES WORKING_DIRECTORY $<TARGET_FILE_DIR:motion-stage>)
# The MSX vendored transport must match the native library API exactly.
foreach(part v9968.c v9968.h geo3d.c geo3d.h)
  file(SHA256 "${LIBV9968_ROOT}/${part}" shared_hash)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/lib/libv9968/${part}" app_hash)
  if(NOT shared_hash STREQUAL app_hash)
    message(FATAL_ERROR "Update vendored libv9968/${part} from LIBV9968_ROOT")
  endif()
endforeach()
if(NOT MSVC)
  target_compile_options(motion-stage PRIVATE -fno-strict-aliasing)
endif()

add_executable(motion-music-test platform/pc/tests/music-trace.c src/music.c)
target_compile_definitions(motion-music-test PRIVATE MOTION_NATIVE)
target_include_directories(motion-music-test PRIVATE ${LIBV9968_ROOT})
add_test(NAME motion-msx-psg-oracle COMMAND motion-music-test)

add_test(NAME motion-native-audio COMMAND motion-stage --frames 180)
set_tests_properties(motion-native-audio PROPERTIES WORKING_DIRECTORY $<TARGET_FILE_DIR:motion-stage> TIMEOUT 15)
